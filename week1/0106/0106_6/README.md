# Урок 6. Инкапсуляция.
## Практическая работа

Разработайте класс с "полной инкапсуляцией", доступ к атрибутам которого и изменение данных реализуются через вызовы методов. В объектно-ориентированном программировании принято имена методов для извлечения данных начинать со слова get (взять), а имена методов, в которых свойствам присваиваются значения, – со слова set (установить). Например, getField, setField.
```PYTHON
class Rectangle:
    """
    Класс, представляющий прямоугольник с методами для установки и получения ширины и высоты.
    """

    def __init__(self, width, height):
        """
        Конструктор класса Rectangle.

        :param width: Ширина прямоугольника.
        :param height: Высота прямоугольника.
        """
        self.w = Rectangle._test(width)  # Устанавливаем ширину, используя статический метод _test для проверки
        self._h = Rectangle._test(height)  # Устанавливаем высоту (с использованием префикса "_", обозначающего защиту), используя статический метод _test для проверки
        print(self)  # Выводим строковое представление прямоугольника при создании

    def setWidth(self, width):
        """
        Устанавливает ширину прямоугольника.

        :param width: Новая ширина прямоугольника.
        """
        self.w = Rectangle._test(width)  # Устанавливаем ширину, используя статический метод _test для проверки

    def setHeight(self, height):
        """
        Устанавливает высоту прямоугольника.

        :param height: Новая высота прямоугольника.
        """
        self._h = Rectangle._test(height)  # Устанавливаем высоту, используя статический метод _test для проверки

    def getWidth(self):
        """
        Возвращает ширину прямоугольника.

        :return: Ширина прямоугольника.
        """
        return self.w  # Возвращаем ширину

    def getHeight(self):
        """
        Возвращает высоту прямоугольника.

        :return: Высота прямоугольника.
        """
        return self._h  # Возвращаем высоту

    @staticmethod
    def _test(value):
        """
        Статический метод для проверки значения.

        Если значение отрицательное, возвращает его абсолютное значение.
        Иначе возвращает само значение.

        :param value: Значение для проверки.
        :return: Проверенное значение (положительное или абсолютное значение).
        """
        if value < 0:
            return abs(value)  # Возвращаем абсолютное значение, если значение отрицательное
        else:
            return value  # Возвращаем значение, если оно положительное

    def __str__(self):
        """
        Возвращает строковое представление прямоугольника.

        :return: Строка в формате "Rectangle {width}x{height}".
        """
        return "Rectangle {0}x{1}".format(self.w, self._h)  # Форматируем и возвращаем строку

# Пример использования
a = Rectangle(3, 4)  # Создаем объект Rectangle с шириной 3 и высотой 4 (выводится строка "Rectangle 3x4")
print(a.getWidth())  # Выводим ширину прямоугольника (3)
a.setWidth(5)  # Устанавливаем ширину равной 5
print(a)  # Выводим строковое представление прямоугольника (выводится строка "Rectangle 5x4")
b = Rectangle(-2, 4)  # Создаем объект Rectangle с шириной -2 и высотой 4 (выводится строка "Rectangle 2x4")  Отрицательная ширина преобразуется в положительную с помощью _test
```
